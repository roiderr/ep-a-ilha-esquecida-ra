// Stop Post Quest
// ============================================================
-	script	William	89,{
	mes "[William]";
	mes "Bem-vindo à";
	mes "Oficina de ^3355FFPlacas^000000 MacMillan.";
	next;
	mes "[William]";
	mes "Minha família, o Clã MacMillan, produz Placas de Trânsito Profissionais há mais de 250 anos. Atualmente, estamos aceitando encomendas especiais para nosso ornamento exclusivo, a ^3355FFPlaca de Pare^000000.";
	next;
	switch(select("Posso pedir uma ^3355FFPlaca de PARE^000000?:Não...")) {
	case 1:
		mes "[William]";
		mes "Este ornamento exclusivo, a ^3355FFPlaca de Pare^000000, é um sinal de trânsito na estrada e também um enfeite de cabelo! Isso, prometemos solenemente com uma garantia de 100 anos.";
		next;
		mes "[William]";
		mes "Para produzir uma ^3355FFPlaca de Pare^000000, precisamos ^00FF0091.100^000000 de zenys, 50 ^00FF00Troncos^000000 e 1 ^00FF00Tintura Preta^000000.";
		next;
		if ((countitem(1019) > 49) && (countitem(983) > 0) && (Zeny > 91099)) {
			delitem 1019,50;  //Wooden_Block
			delitem 983,1;  //Black_Dyestuffs
			set Zeny, Zeny-91100;
			mes "[William]";
			mes "Aqui está!";
			mes "Esta ^3355FFPlaca de Pare^000000 foi feita especialmente para você!";
			mes "Obrigado por parar aqui!";
			mes "...Entendeu a piada?";
			getitem 2272,1; // Stop_Post
			close;
		}
		else {
			mes "[William]";
			mes "Então...";
			mes "Quando você conseguir estes itens, dê uma passada aqui e nós lhe daremos uma Placa de Pare.";
			close;
		}
	case 2:
		mes "[William]";
		mes "De qualquer forma, obrigado por visitar a 'Oficina MacMillan'. Mas pense em comprar algo da próxima vez, ok?";
		close;
	}
}

lhz_in02,230,284,4	script	Funcionário do Hotel#01	86,{
	mes "[Funcionário do Hotel]";
	mes "Bem-vindo ao";
	mes "Dragão Real,";
	mes "onde você vai encontrar";
	mes "as melhores acomodações";
	mes "e os melhores serviços.";
	next;
	switch(select("Salvar:Descansar - 5,000 zeny:Cancelar")) {
	case 1:
		savepoint "lhz_in02",209,275,1,1;
		mes "[Funcionário do Hotel]";
		mes "Muito obrigado, seu";
		mes "Ponto de Retorno foi";
		mes "salvo aqui, no";
		mes "Dragão Real.";
		close;
	case 2:
		if(Zeny < 5000) {
			mes "[Funcionário do Hotel]";
			mes "Desculpe-me, mas você parece";
			mes "não ter zeny o suficiente. Por favor volte quando tiver!";
			close;
		}
		set Zeny,Zeny - 5000;
		mes "[Funcionário do Hotel]";
		mes "Muito obrigado~";
		mes "Espero que você aproveite";
		mes "sua estadia aqui";
		mes "no Dragão Real.";
		close2;
		warp "lhz_in02",219,150;
		percentheal 100,100;
		npcskill "AL_BLESSING",10,99,99;
		end;
	case 3:
		mes "[Funcionário do Hotel]";
		mes "Muito obrigado e";
		mes "tenha um ótimo dia.";
		close;
	}
}

//======================= Inn Function ==============================
function	script	F_InnMaid	{
	.@npc_name$ = getarg(0);

	mes .@npc_name$;
	mes "Bem-vindo à";
	mes getarg(1) + ".";
	mes "Como posso ajudá-lo?";
	next;
	switch(select("Salvar:Descansar - 5.000z:Cancelar")){
	case 1:
		mes .@npc_name$;
		mes "Seu Ponto de Retorno foi salvo.";
		mes "Obrigado e volte sempre.";
		savepoint getarg(2),getarg(3),getarg(4),1,1;
		close;
	case 2:
		mes .@npc_name$;
		if(Zeny < 5000){
			mes "A taxa de serviço é 5.000 zeny.";
			mes "Por favor, tenha certeza de que tem a quantidade necessária para o serviço.";
			close;
		}
		mes "Obrigada.";
		mes "Espero que você aproveite a estadia~";
		close2;
		set Zeny,Zeny - 5000;
		percentheal 100,100;
		return;
	case 3:
		close;
	}
}
